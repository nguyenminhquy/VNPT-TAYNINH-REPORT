
' ================================================
' VBA CODE - DAN VÀO MODULE TRONG VBA EDITOR (Alt+F11)
' ================================================
' Lưu ý: Phải có Python (pandas, openpyxl) và file update_core.py
' trong cùng thư mục với file Excel này

Sub CapNhatBaoCao()
    Dim folderPath As String
    folderPath = ThisWorkbook.Path & "\"
    
    Dim pythonExe As String
    pythonExe = "python"  ' Thay bằng đường dẫn đầy đủ nếu cần, vd: "C:\Python312\python.exe"
    
    Dim scriptPath As String
    scriptPath = folderPath & "update_core.py"
    
    If Dir(scriptPath) = "" Then
        MsgBox "Không tìm thấy file update_core.py trong thư mục:" & Chr(10) & folderPath, vbExclamation, "Lỗi"
        Exit Sub
    End If
    
    Dim result As Integer
    result = MsgBox("Bắt đầu cập nhật báo cáo XLSC?" & Chr(10) & Chr(10) & _
                    "Script sẽ tự động tìm 4 file dữ liệu trong thư mục:" & Chr(10) & folderPath, _
                    vbYesNo + vbQuestion, "Xác nhận cập nhật")
    
    If result = vbNo Then Exit Sub
    
    Dim cmd As String
    Dim logFile As String
    logFile = folderPath & "update_log.txt"
    cmd = pythonExe & " """ & scriptPath & """ > """ & logFile & """ 2>&1"
    
    Application.StatusBar = "Đang cập nhật dữ liệu..."
    Shell "cmd /c " & cmd, vbHide
    Application.Wait Now + TimeValue("00:00:10")
    Application.StatusBar = False
    
    ' Đọc log
    If Dir(logFile) <> "" Then
        Dim fnum As Integer: fnum = FreeFile
        Dim logContent As String: logContent = ""
        Open logFile For Input As #fnum
        Dim ln As String
        Do While Not EOF(fnum): Line Input #fnum, ln: logContent = logContent & ln & Chr(10): Loop
        Close #fnum
        
        If InStr(logContent, "SUCCESS") > 0 Then
            MsgBox "✅ Cập nhật thành công!" & Chr(10) & Chr(10) & _
                   "File kết quả: BaoCao_XLSC_TayNinh_Updated.xlsx" & Chr(10) & _
                   "Vui lòng mở file mới để xem kết quả.", vbInformation, "Thành công"
            ' Mở file kết quả
            Dim outFile As String
            outFile = folderPath & "BaoCao_XLSC_TayNinh_Updated.xlsx"
            If Dir(outFile) <> "" Then Workbooks.Open outFile
        Else
            MsgBox "❌ Có lỗi khi cập nhật:" & Chr(10) & Chr(10) & Left(logContent, 600), vbCritical, "Lỗi"
        End If
    Else
        MsgBox "Đã gửi lệnh cập nhật. Vui lòng chờ 15 giây rồi kiểm tra file BaoCao_XLSC_TayNinh_Updated.xlsx", vbInformation, "Thông báo"
    End If
End Sub
